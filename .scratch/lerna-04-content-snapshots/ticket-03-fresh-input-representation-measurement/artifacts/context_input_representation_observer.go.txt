package decisionfixture

// Owned measurement only: no SQL, body output, context change or authority value.
import (
	"reflect"
	"sync"
	"testing"
	"time"
)

const (
	inputPhaseOther = iota
	inputPhaseCompile
	inputPhaseDispatch
	inputPhaseWorker
	inputPhaseCount
)
const (
	inputRoleObserveInput = iota
	inputRoleBind
	inputRoleBinding
	inputRoleCurrent
	inputRoleAuthorize
	inputRoleStagePublication
	inputRoleRecordFailure
	inputRoleObserve
	inputRoleDispatch
	inputRoleBeginCompile
	inputRoleReserveRead
	inputRoleConfirmRead
	inputRoleCount
)
const (
	inputStageScan = iota
	inputStageClosedJSON
	inputStageJSONBytes
	inputStageDigest
	inputStageFullCompare
	inputStageTupleCompare
	inputStageCount
)

var inputPhaseNames = [inputPhaseCount]string{"other", "compile", "dispatch", "worker"}
var inputRoleNames = [inputRoleCount]string{"observe_input", "bind", "binding", "current", "authorize", "stage_publication", "record_failure", "observe", "dispatch_input", "begin_compile", "reserve_read", "confirm_read"}
var inputStageNames = [inputStageCount]string{"input_scan_sql_wall", "input_closed_json", "input_json_bytes", "input_digest", "full_input_compare", "attempt_tuple_compare"}

type inputRepresentationCell struct {
	Attempted, Completed, Errors, False uint64
	Wall, Max                           time.Duration
	FirstCause                          string
}
type inputRepresentationToken struct {
	phase, role, stage int
	start              time.Time
	valid              bool
}
type inputRepresentationCosts struct {
	mu                 sync.Mutex
	phase, depth       int
	dropped, ambiguity uint64
	cells              [inputPhaseCount][inputRoleCount][inputStageCount]inputRepresentationCell
}

func newInputRepresentationCosts(t *testing.T) *inputRepresentationCosts {
	p := &inputRepresentationCosts{}
	t.Cleanup(func() { p.summary(t) })
	return p
}
func (p *inputRepresentationCosts) enter(phase int) func() {
	if p == nil {
		return func() {}
	}
	p.mu.Lock()
	previous := p.phase
	if phase < 0 || phase >= inputPhaseCount {
		p.dropped++
		phase = inputPhaseOther
	}
	if p.depth != 0 {
		p.ambiguity++
	}
	p.phase = phase
	p.depth++
	p.mu.Unlock()
	return func() {
		p.mu.Lock()
		if p.depth <= 0 || p.phase != phase {
			p.ambiguity++
		} else {
			p.depth--
			p.phase = previous
		}
		p.mu.Unlock()
	}
}
func (p *inputRepresentationCosts) begin(role, stage int) inputRepresentationToken {
	if p == nil {
		return inputRepresentationToken{}
	}
	p.mu.Lock()
	phase := p.phase
	if role < 0 || role >= inputRoleCount || stage < 0 || stage >= inputStageCount || phase < 0 || phase >= inputPhaseCount {
		p.dropped++
		p.mu.Unlock()
		return inputRepresentationToken{}
	}
	if p.depth > 1 {
		p.ambiguity++
	}
	p.cells[phase][role][stage].Attempted++
	p.mu.Unlock()
	return inputRepresentationToken{phase: phase, role: role, stage: stage, start: time.Now(), valid: true}
}
func (p *inputRepresentationCosts) end(token inputRepresentationToken, err error, falseResult bool) {
	if p == nil || !token.valid {
		return
	}
	wall := time.Since(token.start)
	p.mu.Lock()
	c := &p.cells[token.phase][token.role][token.stage]
	c.Completed++
	c.Wall += wall
	if wall > c.Max {
		c.Max = wall
	}
	if falseResult {
		c.False++
	}
	if err != nil {
		c.Errors++
		if c.FirstCause == "" {
			c.FirstCause = claimCause(err)
		}
	}
	p.mu.Unlock()
}
func (p *inputRepresentationCosts) summary(t *testing.T) {
	p.mu.Lock()
	defer p.mu.Unlock()
	t.Logf("INPUT_REP scope=%s slots=288 dropped=%d ambiguity=%d open_phase_depth=%d observer_overhead=UNKNOWN wall_not_CPU=true sql_wall_not_server_time=true", t.Name(), p.dropped, p.ambiguity, p.depth)
	for phase := 0; phase < inputPhaseCount; phase++ {
		for role := 0; role < inputRoleCount; role++ {
			for stage := 0; stage < inputStageCount; stage++ {
				c := p.cells[phase][role][stage]
				if c.Attempted == 0 {
					continue
				}
				cause := c.FirstCause
				if cause == "" {
					cause = "nil"
				}
				t.Logf("INPUT_REP cell phase=%s role=%s stage=%s attempted=%d completed=%d errors=%d false=%d wall_ns=%d max_ns=%d first_cause=%s", inputPhaseNames[phase], inputRoleNames[role], inputStageNames[stage], c.Attempted, c.Completed, c.Errors, c.False, c.Wall.Nanoseconds(), c.Max.Nanoseconds(), cause)
			}
		}
	}
}

func (d *ContextDispatcher) inputRepresentationEqual(role, stage int, left, right any) bool {
	token := d.inputRepresentation.begin(role, stage)
	equal := reflect.DeepEqual(left, right)
	d.inputRepresentation.end(token, nil, !equal)
	return equal
}
func (d *ContextDispatcher) inputRepresentationDigest(role int, body []byte) string {
	token := d.inputRepresentation.begin(role, inputStageDigest)
	value := digest(body)
	d.inputRepresentation.end(token, nil, false)
	return value
}
func (w *ContextWorld) InputRepresentationPhase(phase string) func() {
	switch phase {
	case "compile":
		return w.inputRepresentation.enter(inputPhaseCompile)
	case "dispatch":
		return w.inputRepresentation.enter(inputPhaseDispatch)
	case "worker":
		return w.inputRepresentation.enter(inputPhaseWorker)
	default:
		return w.inputRepresentation.enter(-1)
	}
}
