// Temporary measurement-only overlay: these context labels select counters.
// They contain no authority, ref/body cache, budget, deadline, or current fact.
type ContentQualificationObserver interface {
	BeginContentQualification(role, phase, stage string)
	EndContentQualification(role, phase, stage string, elapsed time.Duration, err error)
}
type contentQualificationKey struct{}
type contentQualificationLabels struct {
	observer    ContentQualificationObserver
	role, phase string
}

func WithContentQualificationObserver(ctx context.Context, observer ContentQualificationObserver, role string) context.Context {
	return context.WithValue(ctx, contentQualificationKey{}, contentQualificationLabels{observer: observer, role: role})
}
func WithContentQualificationRole(ctx context.Context, role string) context.Context {
	if labels, ok := ctx.Value(contentQualificationKey{}).(contentQualificationLabels); ok {
		labels.role = role
		return context.WithValue(ctx, contentQualificationKey{}, labels)
	}
	return ctx
}
func WithContentQualificationPhase(ctx context.Context, phase string) context.Context {
	if labels, ok := ctx.Value(contentQualificationKey{}).(contentQualificationLabels); ok {
		labels.phase = phase
		return context.WithValue(ctx, contentQualificationKey{}, labels)
	}
	return ctx
}
func ContentQualificationActive(ctx context.Context) bool {
	labels, ok := ctx.Value(contentQualificationKey{}).(contentQualificationLabels)
	return ok && labels.observer != nil && labels.phase != ""
}
func BeginContentQualification(ctx context.Context, stage string) time.Time {
	if labels, ok := ctx.Value(contentQualificationKey{}).(contentQualificationLabels); ok && labels.observer != nil && labels.phase != "" {
		labels.observer.BeginContentQualification(labels.role, labels.phase, stage)
		return time.Now()
	}
	return time.Time{}
}
func EndContentQualification(ctx context.Context, stage string, start time.Time, err error) {
	if start.IsZero() {
		return
	}
	if labels, ok := ctx.Value(contentQualificationKey{}).(contentQualificationLabels); ok && labels.observer != nil && labels.phase != "" {
		labels.observer.EndContentQualification(labels.role, labels.phase, stage, time.Since(start), err)
	}
}

var ErrProcessingLimit = errors.New("content processing exceeds remaining byte limit")
var ErrProcessingForbidden = errors.New("content processing forbidden")

// ProcessingRead is an observation of one completed current read. Sources and
// RetainUntil describe that read; they confer no future publication permission.
type ProcessingRead struct {
	Ref         v.ContentRef
	Bytes       []byte
	Sources     []v.ContentRef
	RetainUntil time.Time
}

// ReadForProcessing is an internal consumer seam, separate from public
// disclosure. Both short transactions qualify current read/process for the
// exact target and its complete registered closure; I/O holds no database Tx.
func (s *Service) ReadForProcessing(ctx context.Context, subject *v.SubjectBinding, ref v.ContentRef, purpose string, remaining int64) (ProcessingRead, error) {
	var out ProcessingRead
	principal, ok := trustedSubject(subject, s.config.Owner)
	if !ok || ref.Owner != s.config.Owner || purpose == "" {
		return out, ErrProcessingForbidden
	}
	if !finite(ctx) {
		return out, ErrUnavailable
	}
	if _, err := v.Encode(ref); err != nil {
		return out, err
	}
	length, err := strconv.ParseInt(string(ref.ByteLength), 10, 64)
	if err != nil || length < 0 || length > 262144 || remaining < length {
		return out, ErrProcessingLimit
	}
	var objectKey string
	var before time.Time
	identities := make(closureIdentityMemo, 65)
	observationPhase := "pre"
	observe := func(ctx context.Context, tx runtime.Tx) (observationErr error) {
		ctx = WithContentQualificationPhase(ctx, observationPhase)
		callbackStart := BeginContentQualification(ctx, "callback")
		defer func() { EndContentQualification(ctx, "callback", callbackStart, observationErr) }()
		// Each observation rebuilds its closure. No private records or permission
		// from the first transaction are used as authority by the second one.
		now, err := s.config.Store.Now(ctx, tx)
		if err != nil {
			return err
		}
		p, err := s.config.Store.CheckPolicy(ctx, tx, principal, ref, purpose, []string{"read", "process"}, now)
		if err != nil {
			return err
		}
		if p == nil {
			return ErrProcessingForbidden
		}
		r, err := s.config.Store.LockVersion(ctx, tx, ref)
		if err != nil {
			return err
		}
		now, err = s.config.Store.Now(ctx, tx)
		if err != nil {
			return err
		}
		if !now.Before(p.ValidUntil) || !now.Before(p.RetainUntil) {
			return ErrProcessingForbidden
		}
		actual := ref
		if r != nil {
			actual = r.Ref
		}
		if p.Ref != actual {
			return ErrProcessingForbidden
		}
		if r == nil || r.Ref != ref || r.Publication != "published" {
			return ErrUnavailable
		}
		if r.BodySeal != nil {
			return ErrProcessingForbidden
		}
		bound := earlier(earlier(p.ValidUntil, p.RetainUntil), cutoff(r.CurrentRetainUntil))
		if !now.Before(bound) {
			return ErrProcessingForbidden
		}
		closure, err := s.registeredClosureWithIdentities(ctx, tx, ref, r.Sources, principal, purpose, []string{"read", "process"}, identities)
		if err != nil {
			if _, ok := qualificationCode(err); ok {
				return errors.Join(ErrProcessingForbidden, err)
			}
			return err
		}
		if len(closure.refs) > 0 {
			bound = earlier(bound, earlier(closure.validBefore, closure.retainBefore))
		}
		now, err = s.config.Store.Now(ctx, tx)
		if err != nil {
			return err
		}
		if !now.Before(bound) {
			return ErrProcessingForbidden
		}
		if before.IsZero() {
			before = bound
		} else {
			before = earlier(before, bound)
		}
		objectKey = r.ObjectKey
		out.Sources = append([]v.ContentRef{}, closure.refs...)
		out.RetainUntil = bound
		return nil
	}
	preQualificationContext := WithContentQualificationPhase(ctx, "pre")
	preWithinStart := BeginContentQualification(preQualificationContext, "within")
	if err = s.config.Store.Within(ctx, owner(s.config.Owner), observe); err != nil {
		EndContentQualification(preQualificationContext, "within", preWithinStart, err)
		return ProcessingRead{}, err
	}
	EndContentQualification(preQualificationContext, "within", preWithinStart, nil)
	bounded, cancel := context.WithDeadline(ctx, before)
	defer cancel()
	objectsStart := BeginContentQualification(ctx, "objects.read")
	body, err := s.config.Objects.Read(bounded, objectKey, ref.Hash, length)
	EndContentQualification(ctx, "objects.read", objectsStart, err)
	if err != nil {
		return ProcessingRead{}, err
	}
	integrityStart := BeginContentQualification(ctx, "integrity")
	if err = bounded.Err(); err != nil {
		EndContentQualification(ctx, "integrity", integrityStart, err)
		return ProcessingRead{}, err
	}
	if int64(len(body)) != length || processingHash(body) != ref.Hash {
		EndContentQualification(ctx, "integrity", integrityStart, ErrObjectIntegrity)
		return ProcessingRead{}, ErrObjectIntegrity
	}
	EndContentQualification(ctx, "integrity", integrityStart, nil)
	observationPhase = "post"
	postQualificationContext := WithContentQualificationPhase(ctx, "post")
	postWithinStart := BeginContentQualification(postQualificationContext, "within")
	if err = s.config.Store.Within(ctx, owner(s.config.Owner), observe); err != nil {
		EndContentQualification(postQualificationContext, "within", postWithinStart, err)
		return ProcessingRead{}, err
	}
	EndContentQualification(postQualificationContext, "within", postWithinStart, nil)
	out.Ref = ref
	out.Bytes = body
	return out, nil
}

func processingHash(body []byte) string {
	sum := sha256.Sum256(body)
	return "sha256:" + hex.EncodeToString(sum[:])
}
